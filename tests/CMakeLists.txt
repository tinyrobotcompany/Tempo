cmake_minimum_required(VERSION 3.16)
project(station_host_tests C)
set(CMAKE_C_STANDARD 11)
if(NOT LVGL_SOURCE_DIR)
  message(FATAL_ERROR "Set LVGL_SOURCE_DIR to LVGL 9.5.0")
endif()
if(NOT CJSON_SOURCE_DIR)
  set(CJSON_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../firmware/managed_components/espressif__cjson/cJSON")
endif()
set(LV_CONF_SKIP ON CACHE BOOL "" FORCE)
set(LV_CONF_BUILD_DISABLE_EXAMPLES ON CACHE BOOL "" FORCE)
set(LV_CONF_BUILD_DISABLE_DEMOS ON CACHE BOOL "" FORCE)
add_compile_definitions(LV_USE_LODEPNG=1 LV_USE_QRCODE=1 LV_USE_LOG=1 LV_LOG_PRINTF=1 LV_USE_STDLIB_MALLOC=LV_STDLIB_CLIB LV_CONF_SKIP LV_FONT_MONTSERRAT_18=1 LV_FONT_MONTSERRAT_24=1 LV_FONT_MONTSERRAT_32=1 LV_FONT_MONTSERRAT_48=1)
add_compile_options(-include ${CMAKE_CURRENT_SOURCE_DIR}/test_config.h)
add_subdirectory(${LVGL_SOURCE_DIR} lvgl)
add_executable(test_ui test_ui.c ../firmware/components/station_transport/station_map.c ../firmware/components/station_ui/weather_visuals.c ../firmware/components/station_weather/station_weather.c ${CJSON_SOURCE_DIR}/cJSON.c ../firmware/components/station_ui/fonts/clock_font.c ../firmware/components/station_ui/fonts/light_clock_font.c ../firmware/components/station_ui/fonts/bold_clock_font.c ../firmware/components/station_ui/fonts/hairline_clock_font.c ../firmware/components/station_ui/fonts/condensed_clock_font.c ../firmware/components/station_ui/fonts/wide_clock_font.c ../firmware/components/station_ui/fonts/text_fonts.c ../firmware/components/station_ui/station_ui.c ../firmware/components/station_core/station_core.c ../firmware/components/station_core/timezones.c)
target_include_directories(test_ui PRIVATE ../firmware/components/station_transport/include ../firmware/components/station_weather/include ${CJSON_SOURCE_DIR} ../firmware/components/station_ui/include ../firmware/components/station_core/include)
target_link_libraries(test_ui PRIVATE lvgl m)

add_executable(test_map_tiles test_map_tiles.c)
target_include_directories(test_map_tiles PRIVATE network_stubs ../firmware/main ../firmware/components/station_core/include ../firmware/components/station_transport/include)
target_link_libraries(test_map_tiles PRIVATE lvgl m)

option(TEMPO_COVERAGE "Instrument first-party host test targets" OFF)
if(TEMPO_COVERAGE)
  foreach(target test_ui test_map_tiles)
    target_compile_options(${target} PRIVATE -fprofile-instr-generate -fcoverage-mapping)
    target_link_options(${target} PRIVATE -fprofile-instr-generate)
  endforeach()
endif()
